\begin{ejer}
	Raíces, fracciones y exponentes.
	\begin{displaymath}
		\sqrt[n]{\frac{a}{b}}=
		{\left(\frac{a}{b}\right)}^{\frac{1}{n}}=
		\frac{a^{\frac{1}{n}}}{b^{\frac{1}{n}}}=
		\frac{\sqrt[n]{a}}{\sqrt[n]{b}}.
	\end{displaymath}
\end{ejer}

\begin{ejer}
	Uso del entorno \verb|gather|.
	\begin{gather}
		x^{y^{z}}=a.\\
		x^{2}_{1}=x^{2}_{1}.\\
		x^{2y}=z.\\
		a_{ij}=0.
	\end{gather}
\end{ejer}

\begin{ejer}
	Letras griegas, sombreros y sumatoria.
	\begin{displaymath}
		n^{-1}
		\sum^{n}_{i=1}
		\left(
		y_{i}-
		\hat{\beta}_{0}-
		\hat{\beta}_{1}x_{i}
		\right)=
		0.
	\end{displaymath}

	\begin{displaymath}
		\bar{y}=
		\hat{\beta}_{0}+
		\hat{\beta}_{1}
		\bar{x},
	\end{displaymath}
\end{ejer}
donde $\bar{y}=n^{-1}\sum^{n}_{i=1}y_{i}$ es la muestra promedio de
$y_{i}$ e igualmente por $\bar{x}$.

\begin{ejer}
	Ecuaciones: uso de entorno \verb|equation| y entorno \verb|cases|.
	\begin{equation*}
		\left|x\right| =
		\begin{cases}
			a  & \text{si $a\ge 0$}. \\
			-a & \text{si $a< 0$}.
		\end{cases}
	\end{equation*}
\end{ejer}

\begin{ejer}
	Uso de entorno \verb|equation| y entorno \verb|split|.
	\begin{equation}
		\begin{split}
			a & =b+c-d.          \\
			  & \phantom{=}+e-f. \\
			  & =g+h.            \\
			  & =i.
		\end{split}
	\end{equation}

\begin{ejer}

\begin{ejer}
	Matrices.
	Dado que
	$
		\left|
		\begin{smallmatrix}
			a & h & g\\
			h & b & f\\
			g & f & c
		\end{smallmatrix}
		\right|=0
	$,
	la matriz
	$
		\left(
		\begin{smallmatrix}
			a & h & g\\
			h & b & f\\
			g & f & c
		\end{smallmatrix}
		\right)
	$
	no es invertible.
\end{ejer}
